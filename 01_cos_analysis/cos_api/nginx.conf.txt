events {
    worker_connections 1024;
}

http {
    upstream cos_api {
        server api:5000;
    }

    server {
        listen 80;
        server_name api.cos-kfc.local;
        
        # Redirection vers HTTPS
        return 301 https://$server_name$request_uri;
    }

    server {
        listen 443 ssl http2;
        server_name api.cos-kfc.local;

        # Certificats SSL (à générer)
        ssl_certificate /etc/nginx/ssl/cert.pem;
        ssl_certificate_key /etc/nginx/ssl/key.pem;

        # Configuration SSL
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers HIGH:!aNULL:!MD5;

        # Headers de sécurité
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-XSS-Protection "1; mode=block" always;

        # Logs
        access_log /var/log/nginx/access.log;
        error_log /var/log/nginx/error.log;

        # Taille maximale des requêtes
        client_max_body_size 10M;

        # Location principale
        location / {
            proxy_pass http://cos_api;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            
            # Timeouts
            proxy_connect_timeout 60s;
            proxy_send_timeout 60s;
            proxy_read_timeout 60s;
        }

        # Location pour les métriques Prometheus
        location /metrics {
            proxy_pass http://cos_api/metrics;
            auth_basic "Prometheus Metrics";
            auth_basic_user_file /etc/nginx/.htpasswd;
        }

        # Location pour la santé
        location /health {
            proxy_pass http://cos_api/api/health;
        }
    }
}
